% Preserve the imported PDF's transparency group. Put the driver's additional
% group on a separate outer Form XObject, avoiding conflicting duplicate /Group
% keys. Do not replace this with PDF post-processing or disabling transparency.
% Exact-token patches deliberately fail on an unexpected/new driver: review
% whether the upstream fix is then installed before removing this backport.
\usepackage{etoolbox}
\makeatletter
\newcommand{\QMSBRgraphicsPatchError}[1]{%
  \PackageError{qmsbr-graphics-compat}{Graphics compatibility patch #1 failed}%
    {Expected LuaTeX graphics driver 2025/09/29 v1.2f. Review the upstream fix and local header before rendering.}}
\patchcmd{\Gread@@pdftex}{\ifGread@transgroup 1\fi}{}{}{\QMSBRgraphicsPatchError{1}}
\patchcmd{\Gread@@pdftex}{\ifGread@transgroup/Group<</S/Transparency/K false/I false>>\fi}{}{}{\QMSBRgraphicsPatchError{2}}
\patchcmd{\Gread@@pdftex}
  {\setbox\@tempboxa=\hbox{\useimageresource\lastsavedimageresourceindex}}
  {\setbox\@tempboxa=\hbox{\useimageresource\lastsavedimageresourceindex}%
   \ifGread@transgroup
     \saveboxresource attr{/Group<</S/Transparency/K false/I false>>}\@tempboxa
     \setbox\@tempboxa=\hbox{\useboxresource\lastsavedboxresourceindex}%
   \fi}
  {}{\QMSBRgraphicsPatchError{3}}
\patchcmd{\Gread@@pdftex}
  {\useimageresource\the\lastsavedimageresourceindex}
  {\ifGread@transgroup\useboxresource\the\lastsavedboxresourceindex\else\useimageresource\the\lastsavedimageresourceindex\fi}
  {}{\QMSBRgraphicsPatchError{4}}
\makeatother
